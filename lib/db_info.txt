CREATE TABLE public.users (
  id uuid NOT NULL DEFAULT gen_random_uuid(),
  full_name text NOT NULL,
  phone text,
  email text UNIQUE,
  photo_url text,
  created_at timestamp with time zone DEFAULT now(),
  updated_at timestamp with time zone DEFAULT now(),
  user_no integer NOT NULL UNIQUE,
  password_hash text,
  CONSTRAINT users_pkey PRIMARY KEY (id)
);
CREATE TABLE public.messes (
  id bigint NOT NULL DEFAULT nextval('messes_id_seq'::regclass),

  phone text CHECK (length(phone) <= 20),
  CONSTRAINT messes_pkey PRIMARY KEY (id),
  CONSTRAINT messes_manager_id_fkey FOREIGN KEY (manager_id) REFERENCES public.users(id),
  CONSTRAINT messes_act_manager_id_fkey FOREIGN KEY (act_manager_id) REFERENCES public.users(id)
);
